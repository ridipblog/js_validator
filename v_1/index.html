📦 CDN Module: Js Validator

Validate input field with Js Validator

💡 How to Use

Just paste the following CDN links inside your HTML file:

                    
        

    <script src="https://code.jquery.com/jquery-3.7.1.min.js" integrity="sha256-/JqT3SQfawRcv/BIHPThkBvs0OEvtFFmqPF/lYI/Cxo=" crossorigin="anonymous"></script>

    import ValidateModule from 'https://ridipblog.github.io/js_validator/v_1/ValidatorModule.min.js;

How Implement On HTML Code:

  1. Add `handle-request` class in input if you want validate that input 
                    
    <input type="text" name="user_name" class="handle-request">
    <input type="text" name="user_mail" class="handle-request">
    <input type="text" name="user_phone" class="handle-request">
    <button id="submit-btn">Submit</button>

How Implement on Js Script

                    
    <script>

    // Create New Object of Validator
    const validate_module = new ValidateModule();

    $(document).ready(function () {

        // Add On Click Event By jqery (Or Js)
        $(document).on('click', '#submit-btn', async function () {

            // Add Key - Value Pair , Key is Input Name And Values are Validation area
            const validate_fields = {
                'user_name': ['required'],
                "user_mail": ['required','email'],
                'user_phone':['required','number','phone']
            };

            // Write Custom Message for Validation area
            const messages={
                'required':'plaese enter value for :attribute'
            };

            // Call Validate Function 
            validate_module.validateWithInputs(validate_fields,messages).then((result) => {
                if(result.fail){
                    console.log(result.errors)
                }else{
                }
            }).catch((error) => {
                console.log(error);
            });

            // --------- if direct used with await keyword ------------
            // let validator=await validate_module.validateWithInputs(validate_fields);
        });
    });

    </script>
